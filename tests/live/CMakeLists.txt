cmake_minimum_required(VERSION 3.18)
project(snodec-auth-live-tests LANGUAGES CXX)
find_package(snodec-auth 2 REQUIRED)
add_executable(live-client ../live_client.cpp)
add_executable(live-transport ../live_transport.cpp)
foreach(target live-client live-transport)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_link_libraries(${target} PRIVATE snodec-auth::auth-client)
endforeach()

if(TARGET snodec-auth::auth-idp)
    add_executable(live-idp ../live_idp.cpp)
    target_compile_features(live-idp PRIVATE cxx_std_20)
    target_link_libraries(live-idp PRIVATE snodec-auth::auth-idp)
endif()
